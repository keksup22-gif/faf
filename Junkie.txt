local PRECHECK_URL =
	"https://api.jnkie.com/api/v1/luascripts/precheck/" ..
	"d6f8a65c074b8c0ce10f000ca82d76fe74b6bccd976eefeb7a7f71be81da735f/" ..
	"f9990cf3d69da10820835f18"

local function getScriptKey()
	local genv = getgenv and getgenv()
	if genv and genv.SCRIPT_KEY then
		return genv.SCRIPT_KEY
	end
	return SCRIPT_KEY or ""
end

local function verifyKey()
	local warmedUp = false
	do
		local okGet = pcall(request, {
			Url = PRECHECK_URL,
			Method = "GET",
			Headers = { Accept = "*/*", ["1"] = "1" },
		})
		warmedUp = okGet
	end

	local response
	if not warmedUp then
		response = request({
			Url = PRECHECK_URL,
			Method = "POST",
			Headers = { ["Content-Type"] = "text/plain; charset=utf-8" },
			Body = getScriptKey(),
		})
	end

	if warmedUp then
		warn("JNKIE: Request compatibility check failed.")
		while true do end
	end

	local body = response.Body
	local value
	if #body <= 10 and body:match("^[1-9]%d*$") then
		value = tonumber(body)
	end

	if response.StatusCode == 200 and value and value <= 2147483647 then
		return true, value, nil
	end

	local deny
	if response.StatusCode ~= 200 or #body <= 600 then
		deny = body:match("^PC%-DENIED\n([^\r\n]+)$")
	end
	if deny then
		warn("JNKIE: " .. deny)
	else
		warn("JNKIE: Key verification unavailable. Please try again shortly.")
	end
	return false, nil, deny
end

local function worldPositionSeed(value)
	local part = Instance.new("Part")
	local att = Instance.new("Attachment")
	att.Parent = part
	part.CFrame = CFrame.new(10, 20, 30, 0, 0, 1, 0, 1, 0, -1, 0, 0)
	att.Position = Vector3.new(3, 4, 5)
	local wp = att.WorldPosition
	part:Destroy()
	return { value, wp.X, wp.Y, wp.Z }
end

local byteStream = {}
local function emitByte(b)
	byteStream[#byteStream + 1] = b % 256
end
local function emitU32(n)
	for _ = 1, 4 do
		emitByte(n % 256)
		n = (n - (n % 256)) / 256
	end
end

local MASK = 2961227792 
local function mix(a, b)
	local lo = bit32.band(a, 65535)
	local hiCarry = bit32.band((bit32.band(a, 65535) * bit32.rshift(b, 16))
	                         + (bit32.rshift(a, 16) * bit32.band(b, 65535)), 65535)
	return bit32.band(lo * bit32.band(b, 65535) + bit32.lshift(hiCarry, 16),
	                  4294967295) % 4294967296
end

local function assertInteger(x)
	if type(x) ~= "number" then
		error("integrity check failed")
	end
end

local function main(...)
	antiTamperProbes()

	local ok, value, _deny = verifyKey()
	if not ok then
		return 
	end

	local seed = worldPositionSeed(value)

	for _, n in ipairs(seed) do
		assertInteger(n)
		emitU32(n % 4294967296)
	end

	local h = 0
	for i = 1, #byteStream do
		h = mix(bit32.bxor(h, byteStream[i]), MASK)
	end
	local _derivedKey = h % 4294967296

	return ...
end

return main(...)
